# 2.2 抛物线拟合提取 mean / slope / curve（公式2） -----------------------------------

# ==========================================
# 1. 定义单个向量的线索提取函数
# ==========================================
extract_tone_cues <- function(T_vec) {
  # 确保向量无全 NA 且长度足够
  if (all(is.na(T_vec)) || length(na.omit(T_vec)) < 10) {
    return(tibble(
      mean = NA_real_, slope = NA_real_, curve = NA_real_,
      BLstart = NA_real_, BLslope1 = NA_real_, BLbreakpoint = NA_real_, BLslope2 = NA_real_,
      onglide = NA_real_, offglide = NA_real_, overall = NA_real_
    ))
  }
  
  N <- length(T_vec)
  t <- seq(0, 1, length.out = N)
  
  # ----------------------------------------
  # A. Parabolic Fit (公式 2 移位勒让德多项式)
  # ----------------------------------------
  p1 <- t - 0.5
  p2 <- (t - 0.5)^2 - (1 / 12)
  
  fit_poly <- lm(T_vec ~ p1 + p2)
  coefs_poly <- coef(fit_poly)
  
  c0 <- unname(coefs_poly["(Intercept)"])
  c1 <- unname(coefs_poly["p1"])
  c2 <- unname(coefs_poly["p2"])
  
  # ----------------------------------------
  # B. Broken-line Fit (遍历 d2 线性优化)
  # ----------------------------------------
  best_error <- Inf
  best_params <- NULL
  
  # 遍历断点位置 d2 (0.1 ~ 0.9)
  for (d2_cand in seq(0.1, 0.9, by = 0.01)) {
    x1 <- pmin(t, d2_cand)
    x2 <- pmax(0, t - d2_cand)
    
    fit_bl <- tryCatch(
      lm(T_vec ~ x1 + x2),
      error = function(e) NULL
    )
    
    if (is.null(fit_bl)) next
    
    err <- sum(resid(fit_bl)^2)
    if (err < best_error) {
      best_error <- err
      coefs_bl <- coef(fit_bl)
      best_params <- list(
        d0 = coefs_bl["(Intercept)"],
        d1 = coefs_bl["x1"],
        d2 = d2_cand,
        d3 = coefs_bl["x2"]
      )
    }
  }
  
  if (!is.null(best_params)) {
    d0 <- unname(best_params$d0)
    d1 <- unname(best_params$d1)
    d2 <- unname(best_params$d2)
    d3 <- unname(best_params$d3)
    
    f0_onset  <- d0
    f0_break  <- d0 + d1 * d2
    f0_offset <- d0 + d1 * d2 + d3 * (1 - d2)
    
    # 按照文献定义的差值 (Difference)
    onglide  <- f0_onset - f0_break
    offglide <- f0_break - f0_offset
    overall  <- f0_onset - f0_offset
  } else {
    d0 <- d1 <- d2 <- d3 <- onglide <- offglide <- overall <- NA_real_
  }
  
  # ----------------------------------------
  # C. 返回包含所有参数的单行 Tibble
  # ----------------------------------------
  return(tibble(
    mean = c0,
    slope = c1,
    curve = c2,
    BLstart = d0,
    BLslope1 = d1,
    BLbreakpoint = d2,
    BLslope2 = d3,
    onglide = onglide,
    offglide = offglide,
    overall = overall
  ))
}
